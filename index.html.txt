        <div class="modal-box">
            <h3>发布招工 / 求职</h3>
            <select id="inputType">
                <option value="招工">🔨 我要招人</option>
                <option value="求职">🙋 我要找工作</option>
            </select>
            <input id="inputTitle" placeholder="标题（如：招保洁阿姨 / 找手工活）">
            <input id="inputAge" placeholder="年龄要求（如：35-50岁）">
            <input id="inputContact" placeholder="联系方式（手机 / 微信）">
            <textarea id="inputNote" placeholder="补充说明（选填，如：工作地点、时间）"></textarea>
            <button class="btn-submit" id="saveBtn">✓ 发布</button>
            <button class="btn-close" id="closeModalBtn">取消</button>
        </div>
    </div>

    <script>
        // 读取本地存储的数据
        let jobs = JSON.parse(localStorage.getItem('sheJobs')) || [];
        let currentFilter = 'all';

        // 渲染列表
        function render() {
            const list = document.getElementById('jobList');
            let filtered = jobs;
            if (currentFilter !== 'all') {
                filtered = jobs.filter(j => j.age && j.age.includes(currentFilter.replace('+', '')));
            }

            if (filtered.length === 0) {
                list.innerHTML = '<div class="empty">暂无信息<br>点击下方按钮发布第一条</div>';
                return;
            }

            list.innerHTML = filtered.map(job => `
                <div class="card">
                    <span class="type-badge ${job.type === '招工' ? 'hiring' : 'seeking'}">${job.type === '招工' ? '🔨 招人' : '🙋 求职'}</span>
                    <h3>${job.title}</h3>
                    <div class="meta">
                        ${job.age ? `<span class="tag">👤 ${job.age}</span>` : ''}
                    </div>
                    ${job.note ? `<p style="color:#999;font-size:13px;margin-bottom:8px;">${job.note}</p>` : ''}
                    <span class="contact">📞 ${job.contact}</span>
                    <span class="time">${job.time}</span>
                </div>
            `).join('');
        }

        // 打开弹窗
        document.getElementById('showModalBtn').onclick = () => document.getElementById('myModal').classList.add('show');
        document.getElementById('closeModalBtn').onclick = () => document.getElementById('myModal').classList.remove('show');

        // 发布信息
        document.getElementById('saveBtn').onclick = () => {
            const type = document.getElementById('inputType').value;
            const title = document.getElementById('inputTitle').value.trim();
            const age = document.getElementById('inputAge').value.trim();
            const contact = document.getElementById('inputContact').value.trim();
            const note = document.getElementById('inputNote').value.trim();

            if (!title || !contact) {
                alert('请至少填写标题和联系方式');
                return;
            }

            jobs.unshift({
                type, title, age, contact, note,
                time: new Date().toLocaleString('zh-CN')
            });

            localStorage.setItem('sheJobs', JSON.stringify(jobs));
            render();
            document.getElementById('myModal').classList.remove('show');

            // 清空输入
            document.getElementById('inputTitle').value = '';
            document.getElementById('inputAge').value = '';
            document.getElementById('inputContact').value = '';
            document.getElementById('inputNote').value = '';
        };

        // 年龄筛选
        document.querySelectorAll('.filter-tag').forEach(tag => {
            tag.onclick = () => {
                document.querySelectorAll('.filter-tag').forEach(t => t.classList.remove('active'));
                tag.classList.add('active');
                currentFilter = tag.dataset.age;
                render();
            };
        });

        render();
    </script>
